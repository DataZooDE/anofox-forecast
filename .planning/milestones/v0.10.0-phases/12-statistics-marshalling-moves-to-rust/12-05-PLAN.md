  pub enum CalendarStep { Months(u32), Fixed(i64) }        // Fixed in the input's integer unit (µs for timestamps)
  pub fn detect_step(observed: &[i64]) -> Result<CalendarStep>
      // < 2 distinct observations -> Fixed(86_400_000_000) (one day; preserves ts_forecast_agg's
      // historical default); otherwise mode = detect_frequency(observed); if mode is a whole number of
      // days: 28..=31 -> Months(1), 89..=92 -> Months(3), 365..=366 -> Months(12); else Fixed(mode).
  pub fn future_dates(observed: &[i64], horizon: usize, step: CalendarStep) -> Result<Vec<i64>>
      // observed: µs timestamps (any order, non-empty); base = max(observed).
      // Fixed(s): base + k*s, k = 1..=horizon (s must be > 0).
      // Months(m): month_end = every observed date's day == last day of its month;
      //   target month = base month + k*m (from the base, never cumulative);
      //   day = month_end ? last_day(target) : min(base.day, last_day(target));
      //   time-of-day of base preserved.
  gaps.rs detect_frequency: ties broken by the SMALLER interval, independent of map iteration order
  (e.g. BTreeMap or an explicit (count, Reverse(diff)) key) — doc comment notes the tie rule.
